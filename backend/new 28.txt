let
    Source = #"FullAnalysisConnection",

    // Read CurrentDBASPD from Excel named range
    CurrentDBASPD_raw =
        let
            t = Excel.CurrentWorkbook(){[Name="CurrentDBASPD"]}[Content],
            firstRow = t{0},
            firstCol = Table.ColumnNames(t){0}
        in
            Text.From(Record.Field(firstRow, firstCol)),

    Normalize = (x as any) as text =>
        let
            txt = if x = null then "" else Text.From(x),
            trimmed = Text.Trim(Text.Clean(txt)),
            noNbsp = Text.Replace(trimmed, Character.FromNumber(160), ""),
            lettersOnly = Text.Select(noNbsp, {"A".."Z","a".."z"}),
            upper = Text.Upper(lettersOnly)
        in
            upper,

    CurrentDBASPD = Normalize(CurrentDBASPD_raw),

    WithNorm = Table.AddColumn(Source, "DBASPD_NORM", each Normalize([DBASPD]), type text),

    Filtered = Table.SelectRows(WithNorm, each [DBASPD_NORM] = CurrentDBASPD),

    Final = Table.RemoveColumns(Filtered, {"DBASPD_NORM"})
in
    Final
